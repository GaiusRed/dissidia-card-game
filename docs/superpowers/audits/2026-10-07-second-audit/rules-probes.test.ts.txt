const free = (): Payment => ({ discard: [], dullBackups: [], specialDiscard: null, dullSource: false, sacrificeSource: false, sourceElements: {}, spend: {} });
const pay = (id: string, amount: number, element: 'Fire' | 'Water' = 'Fire'): Payment => ({ ...free(), discard: [id], sourceElements: { [id]: element }, spend: { [element]: amount } });
const pass = (d: ReturnType<typeof driver>) => expect(d.send({ kind: 'pass' }).ok).toBe(true);
const card = (d: ReturnType<typeof driver>, number: string) => Object.values(d.state.cards).find(c => c.card === number)!;

it('A01 observed: Summon gives opponent priority', () => {
  const d = driver({ placements: [{ seat: 0, card: 'P-017R', zone: 'hand' }, { seat: 0, card: 'P-003C', zone: 'field' }, { seat: 0, card: 'P-005R', zone: 'hand' }] });

  expect(d.state.priority).toBe(1);
});
